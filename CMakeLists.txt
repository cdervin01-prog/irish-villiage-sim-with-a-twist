cmake_minimum_required(VERSION 3.20)
project(irish_village VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SQLite3 REQUIRED)
find_package(CURL REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
add_library(village_core src/state.cpp src/memory.cpp src/dialogue.cpp src/simulation.cpp)
target_include_directories(village_core PUBLIC include)
target_link_libraries(village_core PUBLIC SQLite::SQLite3 CURL::libcurl nlohmann_json::nlohmann_json)
target_compile_options(village_core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(village_core PUBLIC VILLAGE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(village src/main.cpp)
target_link_libraries(village PRIVATE village_core)
include(CTest)
if(BUILD_TESTING)
  find_path(DOCTEST_INCLUDE_DIR doctest/doctest.h REQUIRED)
  add_executable(village_tests tests/tests.cpp)
  target_include_directories(village_tests PRIVATE ${DOCTEST_INCLUDE_DIR})
  target_link_libraries(village_tests PRIVATE village_core)
  add_test(NAME village_tests COMMAND village_tests)
endif()
